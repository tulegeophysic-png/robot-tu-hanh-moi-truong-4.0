/* ---------------- LOG ---------------- */
function addLog(message, type = 'info') {
    const logBox = document.getElementById('system-logs');

    if (!logBox) return;

    const logItem = document.createElement('div');
    const timeStr = new Date().toLocaleTimeString();

    logItem.className = 'log-item ' + type;
    logItem.textContent = '[' + timeStr + '] ' + message;

    logBox.prepend(logItem);
}

/* ---------------- POWER ---------------- */
function toggleRobotPower(checkbox) {
    isRobotPowerOn = checkbox.checked;

    const label = document.getElementById('power-status-label');
    const robotStateText = document.getElementById('robot-state-text');

    if (isRobotPowerOn) {
        label.textContent = 'ON (Hoạt động)';
        label.className = 'status-label active';

        robotStateText.textContent = 'Đang đo & quét';
        robotStateText.style.color = '#38bdf8';

        addLog('🟢 [POWER] Robot đã được BẬT nguồn thủ công.', 'info');

        runSystemLoop();
    } else {
        label.textContent = 'OFF (Tắt nguồn)';
        label.className = 'status-label off';

        robotStateText.textContent = 'Đã Tắt (Off)';
        robotStateText.style.color = '#ef4444';

        addLog('🔴 [POWER] Robot đã bị TẮT nguồn thủ công.', 'warn');
    }
}

/* ---------------- CHARGING ---------------- */
function toggleManualCharging() {
    if (!isRobotPowerOn) {
        alert('Vui lòng BẬT nguồn Robot trước!');
        return;
    }

    isManualCharging = !isManualCharging;
    updateChargingUI();
}

function updateChargingUI() {
    const btn = document.getElementById('btn-manual-charge');
    const btnText = document.getElementById('btn-charge-text');
    const robotStateText = document.getElementById('robot-state-text');

    if (isManualCharging || isAutoCharging) {
        btn.classList.add('active');
        btnText.textContent = 'Dừng Sạc & Rời Dock';

        robotStateText.textContent = '🔋 ĐANG Ở TRẠM SẠC';
        robotStateText.style.color = '#eab308';
    } else {
        btn.classList.remove('active');
        btnText.textContent = 'Gọi Robot Về Sạc';

        robotStateText.textContent = 'Đang đo & quét';
        robotStateText.style.color = '#38bdf8';
    }
}

function resetBattery() {
    robotBattery = 100;

    document.getElementById('val-battery').textContent = '100';

    addLog('🔋 [PIN] Đã đặt lại dung lượng Pin về 100%.', 'info');
}

/* ---------------- BATTERY / AUTO DOCKING ---------------- */
function processBatteryLogic() {
    const batCard = document.getElementById('card-battery');
    const batSub = document.getElementById('battery-status-sub');

    if (isManualCharging || isAutoCharging) {

        robotBattery += 10;
        batSub.textContent = 'Trạng thái: Đang sạc ⚡';

        if (robotBattery >= 100) {

            robotBattery = 100;

            isAutoCharging = false;
            isManualCharging = false;

            updateChargingUI();

            addLog(
                '🟢 [AUTO-DOCKING] Pin đã đầy 100%. Robot tự rời Dock tiếp tục nhiệm vụ!',
                'info'
            );
        }

    } else if (isRobotPowerOn) {

        robotBattery -= Math.floor(Math.random() * 4) + 2;

        if (robotBattery < 0) {
            robotBattery = 0;
        }

        batSub.textContent = 'Trạng thái: Đang xả pin';
    }

    document.getElementById('val-battery').textContent = robotBattery;

    if (
        robotBattery <= 15 &&
        !isAutoCharging &&
        !isManualCharging &&
        isRobotPowerOn
    ) {

        batCard.className = 'metric-card battery-critical';

        isAutoCharging = true;

        updateChargingUI();

        addLog(
            '🚨 [AUTO-DOCKING] Pin nguy kịch (' +
            robotBattery +
            '%). Robot tự động về Trạm Sạc!',
            'danger'
        );

    } else if (robotBattery <= 30) {

        batCard.className = 'metric-card battery-low';

    } else {

        batCard.className = 'metric-card';
    }
}

/* ---------------- GAS / FIRE ---------------- */
function processGasAndFireSafety(gasVal, tempVal) {

    const gasCard = document.getElementById('card-gas');
    const gasSub = document.getElementById('gas-status-sub');
    const banner = document.getElementById('emergency-banner');
    const emergencyText = document.getElementById('emergency-text');

    const isGasEmergency = gasVal > 400;
    const isFireEmergency = tempVal > 50.0;

    if (
        (isGasEmergency || isFireEmergency) &&
        !emergencyActive
    ) {

        emergencyActive = true;

        gasCard.classList.add('emergency-alarm');
        banner.classList.remove('hidden');

        let cause;

        if (isGasEmergency) {
            cause = 'RÒ RỈ KHÍ GAS (Đạt ' + gasVal + ' PPM)';
        } else {
            cause = 'NGUY CƠ CHÁY (Nhiệt độ ' + tempVal + '°C)';
        }

        emergencyText.textContent =
            '🚨 CẢNH BÁO KHẨN CẤP: PHÁT HIỆN ' + cause + '!';

        gasSub.textContent = '⚠️ NGUY HIỂM!';
        gasSub.className = 'text-danger';

        addLog(
            '🚨 [CẢNH BÁO GAS/CHÁY] ' + cause,
            'danger'
        );

    } else if (
        !isGasEmergency &&
        !isFireEmergency &&
        emergencyActive
    ) {

        dismissEmergency();
    }
}

function dismissEmergency() {

    emergencyActive = false;

    document
        .getElementById('emergency-banner')
        .classList.add('hidden');

    document
        .getElementById('card-gas')
        .classList.remove('emergency-alarm');

    document
        .getElementById('gas-status-sub')
        .textContent = 'An Toàn';

    document
        .getElementById('gas-status-sub')
        .className = 'text-safe';

    addLog(
        '🟢 [AN TOÀN] Cảnh báo Gas/Cháy đã được tắt hoặc môi trường đã an toàn trở lại.',
        'info'
    );
}

/* ---------------- HEATMAP ---------------- */
function updateHeatmap(pm25Val, co2Val, gasVal) {

    const kitchen = document.getElementById('room-kitchen');
    const kitchenAqi = document.getElementById('aqi-kitchen');

    const danger =
        gasVal > 400 ||
        pm25Val > 55 ||
        co2Val > 1000;

    const warning =
        pm25Val > 35 ||
        co2Val > 800;

    if (danger) {
        kitchen.className = 'room-box danger';
        kitchenAqi.textContent = 'Nguy hiểm';
    } else if (warning) {
        kitchen.className = 'room-box warning';
        kitchenAqi.textContent = 'Cảnh báo';
    } else {
        kitchen.className = 'room-box safe';
        kitchenAqi.textContent = 'Tốt';
    }
}

/* ---------------- ROBOT POSITION ---------------- */
function updateRobotPosition(gasVal, pm25Val, co2Val) {

    const pins = [
        document.getElementById('pin-living'),
        document.getElementById('pin-bed'),
        document.getElementById('pin-kitchen'),
        document.getElementById('pin-office')
    ];

    pins.forEach(function(pin) {
        pin.classList.remove('show');
    });

    let selected;

    if (gasVal > 400 || pm25Val > 55 || co2Val > 1000) {
        selected = document.getElementById('pin-kitchen');
    } else {
        const rooms = [
            'pin-living',
            'pin-bed',
            'pin-kitchen',
            'pin-office'
        ];

        const index =
            Math.floor(Math.random() * rooms.length);

        selected =
            document.getElementById(rooms[index]);
    }

    selected.classList.add('show');
}

/* ---------------- MAIN SENSOR LOOP ---------------- */
function runSystemLoop() {

    if (!isRobotPowerOn) {
        return;
    }

    const tempVal =
        parseFloat(
            (Math.random() * (35 - 25) + 25).toFixed(1)
        );

    const humidityVal =
        parseFloat(
            (Math.random() * (80 - 45) + 45).toFixed(1)
        );

    const pm25Val =
        parseFloat(
            (Math.random() * (100 - 15) + 15).toFixed(1)
        );

    const co2Val =
        parseFloat(
            (Math.random() * (1200 - 400) + 400).toFixed(0)
        );

    let gasVal =
        Math.floor(Math.random() * 200) + 80;

    /* 10% xác suất giả lập rò rỉ Gas */
    if (Math.random() < 0.1) {
        gasVal =
            Math.floor(Math.random() * 300) + 420;
    }

    document.getElementById('val-temp').textContent = tempVal;
    document.getElementById('val-humidity').textContent = humidityVal;
    document.getElementById('val-pm25').textContent = pm25Val;
    document.getElementById('val-co2').textContent = co2Val;
    document.getElementById('val-gas').textContent = gasVal;

    processBatteryLogic();
    processGasAndFireSafety(gasVal, tempVal);

    updateHeatmap(
        pm25Val,
        co2Val,
        gasVal
    );

    updateRobotPosition(
        gasVal,
        pm25Val,
        co2Val
    );

    if (
        !isManualCharging &&
        !isAutoCharging
    ) {

        const now =
            new Date().toLocaleTimeString();

        if (envChart.data.labels.length > 7) {

            envChart.data.labels.shift();

            envChart.data.datasets
                .forEach(function(dataset) {
                    dataset.data.shift();
                });
        }

        envChart.data.labels.push(now);

        envChart.data.datasets[0]
            .data.push(tempVal);

        envChart.data.datasets[1]
            .data.push(gasVal);

        envChart.data.datasets[2]
            .data.push(pm25Val);

        envChart.data.datasets[3]
            .data.push(co2Val);

        envChart.update();
    }
}

/* ---------------- SMART HOME ---------------- */
function toggleDevice(element) {

    element.classList.toggle('active');

    const status =
        element.querySelector('.status-text');

    if (element.classList.contains('active')) {

        status.textContent = 'ON';

        addLog(
            '🟢 [SMART HOME] ' +
            element.querySelector('h4').textContent +
            ' đã BẬT.',
            'info'
        );

    } else {

        status.textContent = 'OFF';

        addLog(
            '⚪ [SMART HOME] ' +
            element.querySelector('h4').textContent +
            ' đã TẮT.',
            'info'
        );
    }
}

/* ---------------- AI VOICE ---------------- */
document
    .getElementById('btn-speak-ai')
    .addEventListener('click', function() {

        const temp =
            document.getElementById('val-temp').textContent;

        const pm25 =
            document.getElementById('val-pm25').textContent;

        const co2 =
            document.getElementById('val-co2').textContent;

        const gas =
            document.getElementById('val-gas').textContent;

        const message =
            'AuraBot AI báo cáo. ' +
            'Nhiệt độ hiện tại ' + temp +
            ' độ C. ' +
            'PM2.5 là ' + pm25 +
            ' microgam trên mét khối. ' +
            'CO2 là ' + co2 +
            ' ppm. ' +
            'Khí gas là ' + gas + ' PPM.';

        if ('speechSynthesis' in window) {

            speechSynthesis.cancel();

            const speech =
                new SpeechSynthesisUtterance(message);

            speech.lang = 'vi-VN';
            speech.rate = 0.95;

            speechSynthesis.speak(speech);

            addLog(
                '🤖 [AI] Trợ lý AI đã đọc báo cáo môi trường.',
                'info'
            );
        }
    });

/* ---------------- EXPORT REPORT ---------------- */
document
    .getElementById('btn-export-excel')
    .addEventListener('click', function() {

        const rows = [
            [
                'Thời gian',
                'Nhiệt độ',
                'Gas',
                'PM2.5',
                'CO2'
            ]
        ];

        for (
            let i = 0;
            i < envChart.data.labels.length;
            i++
        ) {

            rows.push([
                envChart.data.labels[i],
                envChart.data.datasets[0].data[i],
                envChart.data.datasets[1].data[i],
                envChart.data.datasets[2].data[i],
                envChart.data.datasets[3].data[i]
            ]);
        }

        const csv =
            rows
                .map(function(row) {
                    return row.join(',');
                })
                .join('\n');

        const blob =
            new Blob(
                ['\ufeff' + csv],
                {
                    type: 'text/csv;charset=utf-8;'
                }
            );

        const url =
            URL.createObjectURL(blob);

        const link =
            document.createElement('a');

        link.href = url;
        link.download = 'AuraBot_Environment_Report.csv';

        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);

        URL.revokeObjectURL(url);

        addLog(
            '📊 [REPORT] Đã xuất báo cáo dữ liệu môi trường.',
            'info'
        );
    });
